        function handleCustomLogoUpload(event) {
            const file = event.target.files?.[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    const ctx = canvas.getContext('2d');
                    canvas.width = img.width;
                    canvas.height = img.height;
                    ctx.drawImage(img, 0, 0);
                    const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
                    const data = imgData.data;

                    const isWhiteBg = data[0] > 200 && data[1] > 200 && data[2] > 200;
                    let minX = canvas.width, minY = canvas.height, maxX = 0, maxY = 0;

                    for (let y = 0; y < canvas.height; y++) {
                        for (let x = 0; x < canvas.width; x++) {
                            const idx = (y * canvas.width + x) * 4;
                            const r = data[idx], g = data[idx + 1], b = data[idx + 2], a = data[idx + 3];
                            if (isWhiteBg) {
                                const brightness = (r + g + b) / 3;
                                if (brightness < 220 && a > 20) {
                                    if (x < minX) minX = x;
                                    if (x > maxX) maxX = x;
                                    if (y < minY) minY = y;
                                    if (y > maxY) maxY = y;
                                }
                                data[idx] = 255;
                                data[idx + 1] = 255;
                                data[idx + 2] = 255;
                                data[idx + 3] = brightness < 230 ? 255 : 0;
                            } else {
                                const brightness = (r + g + b) / 3;
                                if (brightness > 40 && a > 20) {
                                    if (x < minX) minX = x;
                                    if (x > maxX) maxX = x;
                                    if (y < minY) minY = y;
                                    if (y > maxY) maxY = y;
                                }
                                data[idx] = 255;
                                data[idx + 1] = 255;
                                data[idx + 2] = 255;
                                data[idx + 3] = brightness > 35 ? brightness : 0;
                            }
                        }
                    }

                    ctx.putImageData(imgData, 0, 0);
                    if (maxX > minX && maxY > minY) {
                        const pad = 6;
                        const cropX = Math.max(0, minX - pad);
                        const cropY = Math.max(0, minY - pad);
                        const cropW = Math.min(canvas.width - cropX, (maxX - minX) + pad * 2);
                        const cropH = Math.min(canvas.height - cropY, (maxY - minY) + pad * 2);
                        const trimmed = document.createElement('canvas');
                        trimmed.width = cropW;
                        trimmed.height = cropH;
                        trimmed.getContext('2d').drawImage(canvas, cropX, cropY, cropW, cropH, 0, 0, cropW, cropH);
                        const finalUrl = trimmed.toDataURL('image/png');
                        try { localStorage.setItem('xq_custom_logo', finalUrl); } catch (err) {}
                        applyCustomLogo(finalUrl);
                    } else {
                        applyCustomLogo(e.target.result);
                    }
                };
                img.src = e.target.result;
            };
            reader.readAsDataURL(file);

        function renderCategoryFields(category) {
            document.getElementById('dynamicFieldsContainer').innerHTML = categoryTemplates[category] || categoryTemplates.oem;
        }

        function toggleAeroOtherInput() {
            const checkbox = document.getElementById('aeroOtherCheckbox');
            const wrapper = document.getElementById('aeroOtherWrapper');
            if (checkbox && wrapper) {
                if (checkbox.checked) {
                    wrapper.classList.remove('hidden');
                    document.getElementById('aeroOtherCustom')?.focus();
                } else {
                    wrapper.classList.add('hidden');
                }
            }
        }

        // Initialize default category on load
        renderCategoryFields('oem');

        // Helper when clicking service card links
        function selectCategory(category) {
            const radio = document.querySelector(`input[name="category"][value="${category}"]`);
            if (radio) {
                radio.checked = true;
                renderCategoryFields(category);
            }
        }

        // Generate Formatted LINE Quote Text & Open Modal
        function generateLineQuote(e) {
            e.preventDefault();

            const category = document.querySelector('input[name="category"]:checked').value;
            const make = document.getElementById('carMake').value.trim();
            const model = document.getElementById('carModel').value.trim();
            const year = document.getElementById('carYear').value.trim();
            const body = document.getElementById('carBody').value;
            const notes = document.getElementById('carNotes').value.trim() || '無';

            let serviceTitle = '';
            let detailsLines = [];

            if (category === 'oem') {
                serviceTitle = '❶ 原廠強化系列';
                const checkedItems = Array.from(document.querySelectorAll('input[name="oemItems"]:checked')).map(el => el.value);
                const rotor = document.getElementById('oemRotor').value;
                const pad = document.getElementById('oemPad').value;
                detailsLines.push(`・強化項目：${checkedItems.length ? checkedItems.join('、') : '請專員推薦'}`);
                detailsLines.push(`・碟盤樣式：${rotor}`);
                detailsLines.push(`・來令片材質：${pad}`);
            } else if (category === 'brake') {
                serviceTitle = '❷ 高階 / 碳陶瓷煞車';
                const brand = document.getElementById('brakeBrand').value;
                const axle = document.getElementById('brakeAxle').value;
                const rotor = document.getElementById('brakeRotor').value;
                const pad = document.getElementById('brakePad').value;
                detailsLines.push(`・品牌偏好：${brand}`);
                detailsLines.push(`・安裝位置：${axle}`);
                detailsLines.push(`・碟盤樣式：${rotor}`);
                detailsLines.push(`・來令片材質：${pad}`);
            } else if (category === 'exhaust') {
                serviceTitle = '❸ 精品排氣系統';
                const scope = document.getElementById('exhaustScope').value;
                const bumper = document.getElementById('exhaustBumper').value;
                detailsLines.push(`・排氣升級範圍：${scope}`);
                detailsLines.push(`・後保桿/排氣狀態：${bumper}`);
            } else if (category === 'aero') {
                serviceTitle = '❹ 外觀空力套件';
                const material = document.getElementById('aeroMaterial').value;
                const parts = Array.from(document.querySelectorAll('input[name="aeroParts"]:checked')).map(el => {
                    if (el.value === '其他') {
                        const customPart = document.getElementById('aeroOtherCustom')?.value.trim();
                        return customPart ? `其他（${customPart}）` : '其他';
                    }
                    return el.value;
                });
                detailsLines.push(`・材質偏好：${material}`);
                detailsLines.push(`・欲改裝部位：${parts.length ? parts.join('、') : '全車空力評估'}`);
            } else if (category === 'trading') {
                serviceTitle = '❺ 買車 / 賣車與綜合服務';
                const tType = document.getElementById('tradingType').value;
                const tBrief = document.getElementById('tradingBrief').value.trim();
                detailsLines.push(`・服務需求：${tType}`);
                if (tBrief) detailsLines.push(`・需求簡述：${tBrief}`);
            }

            const formattedText = [
                `【XQ PERFORMANCE 專屬升級諮詢】`,
                `車款/年份：${year} ${make} ${model} (${body})`,
                `升級需求：${serviceTitle}`,
                ...detailsLines,
                `補充說明：${notes}`
            ].join('\n');

            document.getElementById('generatedQuoteText').value = formattedText;
            document.getElementById('copyBtnLabel').innerText = '一鍵複製諮詢內容';
            document.getElementById('quoteModal').classList.remove('hidden');
        }

        // Copy to Clipboard via execCommand (Iframe safe)
        function copyQuoteText() {
            const textarea = document.getElementById('generatedQuoteText');
            textarea.select();
            textarea.setSelectionRange(0, 99999);
            try {
                document.execCommand('copy');
                const label = document.getElementById('copyBtnLabel');
                label.innerText = '已複製！請點擊下方按鈕前往 LINE 貼上';
                setTimeout(() => {
                    label.innerText = '一鍵複製諮詢內容';
                }, 3500);
            } catch (err) {
                console.error('Copy failed', err);
            }
        }

        function closeModal() {
            document.getElementById('quoteModal').classList.add('hidden');
        }

        // Filterable Showcase Gallery
        function filterGallery(filter, btnEl) {
            document.querySelectorAll('.filter-btn').forEach(btn => {
                btn.classList.remove('bg-brand-red', 'text-white', 'border-brand-red');
                btn.classList.add('bg-brand-card', 'text-gray-300', 'border-brand-border');
            });
            btnEl.classList.remove('bg-brand-card', 'text-gray-300', 'border-brand-border');
            btnEl.classList.add('bg-brand-red', 'text-white', 'border-brand-red');

            document.querySelectorAll('.gallery-card').forEach(card => {
                if (filter === 'all' || card.classList.contains(filter)) {
                    card.style.display = 'block';
                } else {
                    card.style.display = 'none';
                }
            });
        }
